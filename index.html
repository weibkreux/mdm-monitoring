How Real Google Sheet Sync Works:

  1. Open your target Google Sheet. Go to Extensions > Apps Script.
  2. Paste the snippet below into `Code.gs` and click Deploy > New Deployment.
  3. Select Web app, set access to Anyone, and deploy.
  4. Paste your Web App URL into the configuration input field below!
Google Apps Script Snippet (`Code.gs`) 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Add header if empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(data.headers);
    }
    
    // Append row data
    sheet.appendRow(data.rowValues);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var action = e.parameter ? e.parameter.action : '';
  var callback = e.parameter ? e.parameter.callback : null;
  var activeUser = Session.getActiveUser().getEmail() || Session.getEffectiveUser().getEmail();
  var result = JSON.stringify({ email: activeUser });

  if (callback) {
    return ContentService.createTextOutput(callback + '(' + result + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(result)
    .setMimeType(ContentService.MimeType.JSON);
}
